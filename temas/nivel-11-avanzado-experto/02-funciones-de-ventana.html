Funciones de Ventana

Las funciones de ventana (window functions) realizan cálculos sobre un conjunto de filas relacionado con la fila actual, sin agrupar el resultado. Son esenciales para reportes, rankings y análisis de series temporales.

Rendimiento superior

Las funciones de ventana suelen ser más eficientes que las subconsultas correlacionadas o GROUP BY cuando necesitas mantener las filas individuales en el resultado.

Sintaxis General

T-SQL
SELECT columna,
       FUNCION_VENTANA() OVER (
           PARTITION BY columna_part
           ORDER BY columna_order
           ROWS/RANGE BETWEEN ... AND ...
       ) AS alias
FROM tabla;

Funciones de Ranking

T-SQL
SELECT Producto, Categoria, Precio,
       ROW_NUMBER() OVER (ORDER BY Precio DESC) AS RowNum,
       RANK()       OVER (ORDER BY Precio DESC) AS Rank,
       DENSE_RANK() OVER (ORDER BY Precio DESC) AS DenseRank,
       NTILE(4)     OVER (ORDER BY Precio DESC) AS Cuartil
FROM Productos;

Funciones de Desplazamiento

T-SQL
-- LAG: fila anterior, LEAD: fila siguiente
SELECT Fecha, Ventas,
       LAG(Ventas, 1)  OVER (ORDER BY Fecha) AS VentaAnterior,
       LEAD(Ventas, 1) OVER (ORDER BY Fecha) AS VentaSiguiente,
       Ventas - LAG(Ventas, 1) OVER (ORDER BY Fecha) AS Diferencias
FROM VentasDiarias;

Funciones Agregadas con OVER

T-SQL
SELECT Departamento, Empleado, Salario,
       SUM(Salario)   OVER (PARTITION BY Departamento) AS TotalDepto,
       AVG(Salario)   OVER (PARTITION BY Departamento) AS PromedioDepto,
       MAX(Salario)   OVER (PARTITION BY Departamento) AS MaxDepto,
       MIN(Salario)   OVER (PARTITION BY Departamento) AS MinDepto,
       COUNT(*)       OVER (PARTITION BY Departamento) AS EmpleadosDepto,
       Salario * 1.0 / SUM(Salario) OVER (PARTITION BY Departamento) * 100
           AS PorcentajeDepto
FROM Empleados;

Ventanas Acumuladas

T-SQL
-- Running total con ROWS
SELECT Fecha, Ventas,
       SUM(Ventas) OVER (
           ORDER BY Fecha
           ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
       ) AS VentasAcumuladas,
       AVG(Ventas) OVER (
           ORDER BY Fecha
           ROWS BETWEEN 6 PRECEDING AND CURRENT ROW
       ) AS MediaMovil7Dias
FROM VentasDiarias;
