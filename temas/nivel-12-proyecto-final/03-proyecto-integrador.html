Proyecto Integrador: Sistema de Gestión de Biblioteca

Arquitectura del Sistema de Gestión de Biblioteca
Expansión del proyecto

Este proyecto puede extenderse agregando multas, reservas, notificaciones por email, reportes estadísticos con window functions, o una API REST con Node.js/PHP.

Requisitos del Sistema

Gestión de libros, autores, miembros y préstamos de una biblioteca. Debe soportar consultas, registro de préstamos, devoluciones, y auditoría de cambios.

DDL — Creación de Tablas

T-SQL
CREATE TABLE dbo.Books (
    BookID      INT IDENTITY(1,1) PRIMARY KEY,
    Title       NVARCHAR(200) NOT NULL,
    ISBN        NVARCHAR(20) UNIQUE NOT NULL,
    Category    NVARCHAR(100),
    PublishedYear INT,
    CopiesTotal INT NOT NULL DEFAULT 1,
    CopiesAvailable INT NOT NULL DEFAULT 1,
    CreatedAt   DATETIME2 DEFAULT GETDATE()
);

CREATE TABLE dbo.Authors (
    AuthorID    INT IDENTITY(1,1) PRIMARY KEY,
    FirstName   NVARCHAR(100) NOT NULL,
    LastName    NVARCHAR(100) NOT NULL,
    Nationality NVARCHAR(100)
);

CREATE TABLE dbo.BookAuthors (
    BookID   INT NOT NULL REFERENCES dbo.Books(BookID),
    AuthorID INT NOT NULL REFERENCES dbo.Authors(AuthorID),
    PRIMARY KEY (BookID, AuthorID)
);

CREATE TABLE dbo.Members (
    MemberID    INT IDENTITY(1,1) PRIMARY KEY,
    FirstName   NVARCHAR(100) NOT NULL,
    LastName    NVARCHAR(100) NOT NULL,
    Email       NVARCHAR(200) UNIQUE NOT NULL,
    Phone       NVARCHAR(20),
    MemberSince DATE DEFAULT GETDATE(),
    IsActive    BIT DEFAULT 1
);

CREATE TABLE dbo.Loans (
    LoanID      INT IDENTITY(1,1) PRIMARY KEY,
    BookID      INT NOT NULL REFERENCES dbo.Books(BookID),
    MemberID    INT NOT NULL REFERENCES dbo.Members(MemberID),
    LoanDate    DATE NOT NULL DEFAULT GETDATE(),
    DueDate     DATE NOT NULL,
    ReturnDate  DATE NULL,
    Status      VARCHAR(20) DEFAULT 'Active'
        CHECK (Status IN ('Active', 'Returned', 'Overdue')),
    CreatedAt   DATETIME2 DEFAULT GETDATE()
);

-- Índices
CREATE INDEX IX_Loans_MemberID ON dbo.Loans(MemberID);
CREATE INDEX IX_Loans_BookID   ON dbo.Loans(BookID);
CREATE INDEX IX_Loans_Status   ON dbo.Loans(Status) WHERE Status = 'Active';
CREATE INDEX IX_Books_Category ON dbo.Books(Category);

Consultas

T-SQL
-- Libros con sus autores
SELECT b.Title, b.ISBN,
       STRING_AGG(a.LastName + ', ' + a.FirstName, '; ') AS Authors
FROM dbo.Books b
JOIN dbo.BookAuthors ba ON b.BookID = ba.BookID
JOIN dbo.Authors a ON ba.AuthorID = a.AuthorID
GROUP BY b.Title, b.ISBN;

-- Préstamos activos con info del miembro
SELECT l.LoanID, m.FirstName + ' ' + m.LastName AS Member,
       b.Title, l.LoanDate, l.DueDate,
       DATEDIFF(DAY, l.DueDate, GETDATE()) AS DaysOverdue
FROM dbo.Loans l
JOIN dbo.Members m ON l.MemberID = m.MemberID
JOIN dbo.Books b   ON l.BookID = b.BookID
WHERE l.Status = 'Active';

Stored Procedure: Registrar Préstamo

T-SQL
CREATE PROCEDURE dbo.usp_BorrowBook
    @BookID   INT,
    @MemberID INT,
    @LoanDays INT = 14
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @Available INT;

    SELECT @Available = CopiesAvailable
    FROM dbo.Books WITH (UPDLOCK, ROWLOCK)
    WHERE BookID = @BookID;

    IF @Available IS NULL
        THROW 50001, 'Libro no encontrado.', 1;

    IF @Available <= 0
        THROW 50002, 'No hay copias disponibles.', 1;

    BEGIN TRY
        BEGIN TRANSACTION;

        INSERT INTO dbo.Loans (BookID, MemberID, LoanDate, DueDate)
        VALUES (@BookID, @MemberID, GETDATE(), DATEADD(DAY, @LoanDays, GETDATE()));

        UPDATE dbo.Books
        SET CopiesAvailable = CopiesAvailable - 1
        WHERE BookID = @BookID;

        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH
        ROLLBACK TRANSACTION;
        THROW;
    END CATCH
END;

Stored Procedure: Devolver Libro

T-SQL
CREATE PROCEDURE dbo.usp_ReturnBook
    @LoanID INT
AS
BEGIN
    SET NOCOUNT ON;

    UPDATE dbo.Loans
    SET ReturnDate = GETDATE(),
        Status = 'Returned'
    WHERE LoanID = @LoanID
      AND Status = 'Active';

    IF @@ROWCOUNT = 0
        THROW 50003, 'Préstamo no encontrado o ya devuelto.', 1;

    UPDATE b
    SET CopiesAvailable = CopiesAvailable + 1
    FROM dbo.Books b
    JOIN dbo.Loans l ON b.BookID = l.BookID
    WHERE l.LoanID = @LoanID;
END;

Vista: Préstamos Activos

T-SQL
CREATE VIEW dbo.v_ActiveLoans AS
SELECT l.LoanID,
       m.FirstName + ' ' + m.LastName AS Member,
       b.Title AS Book,
       l.LoanDate, l.DueDate,
       DATEDIFF(DAY, l.DueDate, GETDATE()) AS OverdueDays
FROM dbo.Loans l
JOIN dbo.Members m ON l.MemberID = m.MemberID
JOIN dbo.Books b   ON l.BookID = b.BookID
WHERE l.Status = 'Active';

Trigger: Auditoría de Préstamos

T-SQL
CREATE TABLE dbo.LoanAudit (
    AuditID     INT IDENTITY PRIMARY KEY,
    LoanID      INT,
    ActionType  VARCHAR(20),
    OldStatus   VARCHAR(20),
    NewStatus   VARCHAR(20),
    ChangedBy   NVARCHAR(128),
    ChangedAt   DATETIME2 DEFAULT GETDATE()
);

CREATE TRIGGER dbo.trg_LoanAudit
ON dbo.Loans
AFTER UPDATE
AS
BEGIN
    SET NOCOUNT ON;

    INSERT INTO dbo.LoanAudit (LoanID, ActionType, OldStatus, NewStatus, ChangedBy)
    SELECT i.LoanID, 'STATUS_CHANGE', d.Status, i.Status, SYSTEM_USER
    FROM inserted i
    JOIN deleted d ON i.LoanID = d.LoanID
    WHERE i.Status <> d.Status;
END;
