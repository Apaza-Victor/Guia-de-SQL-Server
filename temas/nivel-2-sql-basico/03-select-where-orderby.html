SELECT, WHERE, ORDER BY

El comando SELECT es la instrucción más utilizada en SQL. Permite consultar y recuperar datos de una o más tablas.

Sintaxis básica

T-SQL 
SELECT columna1, columna2, ...
FROM nombre_tabla
WHERE condicion
ORDER BY columna [ASC | DESC];

Ejemplo con tabla Products

T-SQL 
-- Crear tabla de ejemplo
CREATE TABLE Products (
    ProductID INT IDENTITY(1,1) PRIMARY KEY,
    Name NVARCHAR(100) NOT NULL,
    Category NVARCHAR(50),
    Price DECIMAL(10,2),
    Stock INT,
    Active BIT DEFAULT 1
);

INSERT INTO Products (Name, Category, Price, Stock)
VALUES
    ('Laptop', 'Electrónica', 1200.00, 15),
    ('Mouse', 'Electrónica', 25.50, 100),
    ('Teclado', 'Electrónica', 45.00, 60),
    ('Silla', 'Muebles', 250.00, 20),
    ('Escritorio', 'Muebles', 450.00, 8),
    ('Monitor', 'Electrónica', 350.00, 30);

-- SELECT *
SELECT * FROM Products;

-- SELECT con columnas específicas
SELECT Name, Price FROM Products;

-- WHERE con operadores
SELECT Name, Price, Stock
FROM Products
WHERE Price > 100;

ORDER BY

T-SQL 
-- Orden ascendente (predeterminado)
SELECT Name, Price FROM Products
ORDER BY Price;

-- Orden descendente
SELECT Name, Price FROM Products
ORDER BY Price DESC;

-- Orden por múltiples columnas
SELECT Category, Name, Price FROM Products
ORDER BY Category ASC, Price DESC;

DISTINCT y TOP

T-SQL 
-- DISTINCT: valores únicos
SELECT DISTINCT Category FROM Products;

-- TOP: limitar filas
SELECT TOP 3 Name, Price FROM Products
ORDER BY Price DESC;

-- TOP con porcentaje
SELECT TOP 50 PERCENT Name, Price FROM Products
ORDER BY Price DESC;

-- TOP WITH TIES (incluye empates)
SELECT TOP 3 WITH TIES Name, Price FROM Products
ORDER BY Price DESC;
Nota

La cláusula TOP WITH TIES incluye filas adicionales que empatan en el último valor. Es útil cuando quieres los N mejores pero no quieres excluir elementos con el mismo valor.
