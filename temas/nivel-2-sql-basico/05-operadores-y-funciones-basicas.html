Operadores y funciones básicas

SQL Server ofrece una amplia gama de operadores y funciones para filtrar y transformar datos.

Operadores de filtrado

LIKE

Permite buscar patrones en cadenas de texto usando comodines.

T-SQL 
-- % = cualquier secuencia de caracteres
SELECT Name FROM Products WHERE Name LIKE 'L%';  -- Empieza con L
SELECT Name FROM Products WHERE Name LIKE '%to'; -- Termina en "to"
SELECT Name FROM Products WHERE Name LIKE '%ap%'; -- Contiene "ap"

-- _ = un solo carácter
SELECT Name FROM Products WHERE Name LIKE 'M__se'; -- "Mouse", 5 letras

-- [ ] = rango o lista
SELECT Name FROM Products WHERE Name LIKE '[A-M]%'; -- De la A a la M

IN, BETWEEN, IS NULL

T-SQL 
-- IN: múltiples valores
SELECT Name, Category FROM Products
WHERE Category IN ('Electrónica', 'Muebles');

-- BETWEEN: rango inclusivo
SELECT Name, Price FROM Products
WHERE Price BETWEEN 50 AND 500;

-- IS NULL: valores nulos
SELECT Name FROM Products
WHERE Description IS NULL;

-- IS NOT NULL: valores no nulos
SELECT Name FROM Products
WHERE Description IS NOT NULL;

Funciones de cadena

T-SQL 
SELECT
    UPPER('hola mundo') AS Mayusculas,
    LOWER('HOLA MUNDO') AS Minusculas,
    LEN('SQL Server') AS Longitud,
    SUBSTRING('SQL Server', 1, 3) AS Subcadena,
    TRIM('  texto  ') AS SinEspacios,
    REPLACE('Hola Mundo', 'Mundo', 'SQL') AS Reemplazo,
    CONCAT('SQL', ' ', 'Server') AS Concatenacion,
    LEFT('SQL Server', 3) AS Izquierda,
    RIGHT('SQL Server', 6) AS Derecha;

Funciones de fecha

T-SQL 
SELECT
    GETDATE() AS FechaActual,
    DATEADD(DAY, 30, GETDATE()) AS DentroDe30Dias,
    DATEDIFF(DAY, '2024-01-01', GETDATE()) AS DiasDesdeEnero,
    YEAR(GETDATE()) AS Anio,
    MONTH(GETDATE()) AS Mes,
    DAY(GETDATE()) AS Dia,
    EOMONTH(GETDATE()) AS FinDeMes,
    FORMAT(GETDATE(), 'dd/MM/yyyy') AS FechaFormateada;

Funciones agregadas (introducción)

T-SQL 
SELECT
    COUNT(*) AS TotalProductos,
    COUNT(DISTINCT Category) AS CategoriasUnicas,
    SUM(Stock) AS StockTotal,
    AVG(Price) AS PrecioPromedio,
    MIN(Price) AS PrecioMinimo,
    MAX(Price) AS PrecioMaximo
FROM Products;
