Conexión con .NET y Java

Tanto .NET como Java tienen soporte maduro y nativo para SQL Server. Veamos ejemplos prácticos con Microsoft.Data.SqlClient y mssql-jdbc.

using() en C#

Siempre usa el bloque using para SqlConnection y SqlCommand. Garantiza que los recursos se liberen incluso si ocurre una excepción.

.NET (C#) con SqlClient

C#
using Microsoft.Data.SqlClient;

string connStr = "Server=localhost;Database=MiBase;User Id=sa;Password=ContraseñaSegura;Encrypt=True;TrustServerCertificate=True;";

string email = "user@example.com";

await using var conn = new SqlConnection(connStr);
await conn.OpenAsync();

await using var cmd = new SqlCommand(
    "SELECT * FROM Usuarios WHERE email = @email", conn);
cmd.Parameters.AddWithValue("@email", email);

await using var reader = await cmd.ExecuteReaderAsync();
while (await reader.ReadAsync())
{
    Console.WriteLine($"{reader["nombre"]} - {reader["email"]}");
}

Transacción en .NET

C#
await using var conn = new SqlConnection(connStr);
await conn.OpenAsync();

await using var tx = conn.BeginTransaction();
try
{
    await using var cmd1 = new SqlCommand(
        "INSERT INTO Log (msg) VALUES ('Inicio')", conn, tx);
    await cmd1.ExecuteNonQueryAsync();

    await using var cmd2 = new SqlCommand(
        "UPDATE Cuenta SET saldo = saldo - 100 WHERE id = 1", conn, tx);
    await cmd2.ExecuteNonQueryAsync();

    await tx.CommitAsync();
}
catch
{
    await tx.RollbackAsync();
    throw;
}

Java con JDBC (mssql-jdbc)

Java
import java.sql.*;

public class SqlServerExample {
    public static void main(String[] args) {
        String connStr = "jdbc:sqlserver://localhost:1433;"
            + "databaseName=MiBase;"
            + "encrypt=true;trustServerCertificate=true;";

        try (Connection conn = DriverManager.getConnection(
                connStr, "sa", "ContraseñaSegura")) {

            String sql = "SELECT * FROM Usuarios WHERE email = ?";
            try (PreparedStatement pstmt = conn.prepareStatement(sql)) {
                pstmt.setString(1, "user@example.com");

                try (ResultSet rs = pstmt.executeQuery()) {
                    while (rs.next()) {
                        System.out.println(
                            rs.getString("nombre") + " - "
                            + rs.getString("email"));
                    }
                }
            }
        } catch (SQLException e) {
            e.printStackTrace();
        }
    }
}
